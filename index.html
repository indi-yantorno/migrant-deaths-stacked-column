<script>

// Cases per cause (rows) by month, Jan to Dec (columns). Used right away,

// then replaced by numbers counted from the CSV if the file can be loaded.

var EMBEDDED = [[4,4,6,7,26,85,98,48,22,20,5,5],[0,1,11,8,3,39,18,15,13,4,2,1],[6,6,4,4,4,6,8,12,11,13,14,6],[6,3,4,9,2,8,8,4,4,5,10,5],[1,1,6,2,4,4,1,2,2,2,6,3],[2,3,4,3,3,14,8,14,2,7,2,3]];


var CAUSES = ['Environmental exposure', 'Water-related', 'Skeletal remains', 'Motor vehicle', 'Fall', 'Other or unknown'];

var MONTHS = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];

var MONTH_FULL = ['January','February','March','April','May','June','July','August','September','October','November','December'];

var CSV_URLS = [

  'deaths.csv',

  encodeURI('https://sandboxtutor.biz/udata/XG68El Paso Sector Migrant Deaths 2008-2023 - Sheet1.csv'),

  encodeURI('https://sandboxtutor.biz/udata/XG68/El Paso Sector Migrant Deaths 2008-2023 - Sheet1.csv')

];


/* ---------- CSV loading ---------- */

function parseCSV(text) {

  var rows = [], row = [], f = '', q = false;

  for (var i = 0; i < text.length; i++) {

    var c = text[i];

    if (q) { if (c === '"' && text[i+1] === '"') { f += '"'; i++; } else if (c === '"') q = false; else f += c; }

    else if (c === '"') q = true;

    else if (c === ',') { row.push(f); f = ''; }

    else if (c === '\n' || c === '\r') { if (c === '\r' && text[i+1] === '\n') i++; row.push(f); rows.push(row); row = []; f = ''; }

    else f += c;

  }

  if (f || row.length) { row.push(f); rows.push(row); }

  return rows;

}

function norm(s) { return String(s || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().replace(/-/g, ' ').trim(); }

var CA = { 'environmental exposure': 0, 'water related': 1, 'skeletal': 2, 'motor vehicle related': 3, 'fall': 4 };


function toData(text) {

  var rows = parseCSV(text.trim()), h = rows[0].map(function (s) { return s.trim(); }), ix = {};

  ['Case Number','Year','Month','Type of Incident'].forEach(function (k) { ix[k] = h.indexOf(k); if (ix[k] < 0) throw new Error('Missing column ' + k); });

  var month = d3.range(6).map(function () { return d3.range(12).map(function () { return 0; }); }), n = 0;

  rows.slice(1).forEach(function (r) {

    var y = parseInt(r[ix['Year']], 10);

    if (!r[ix['Case Number']] || !r[ix['Case Number']].trim() || isNaN(y) || y < 2008 || y > 2024) return;

    var ca = CA[norm(r[ix['Type of Incident']])]; if (ca === undefined) ca = 5;

    var mo = MONTH_FULL.indexOf(String(r[ix['Month']] || '').trim().replace(/^./, function (c) { return c.toUpperCase(); }));

    if (mo >= 0) { month[ca][mo]++; n++; }

  });

  if (n < 100) throw new Error('Too few rows');

  return month;

}

function loadData(i) {

  if (i >= CSV_URLS.length) return Promise.reject(new Error('No CSV link worked'));

  return fetch(CSV_URLS[i]).then(function (r) { if (!r.ok) throw new Error(r.status); return r.text(); }).then(toData)

    .catch(function () { return loadData(i + 1); });

}


/* ---------- Legend (click a cause to hide or show it) ---------- */

var data = EMBEDDED, off = {};

var tip = d3.select('#tip'), wrap = document.getElementById('wrap');

CAUSES.forEach(function (label, k) {

  d3.select('#legend').append('button').attr('type', 'button').attr('aria-pressed', 'true')

    .html('<i class="sw" style="background:var(--c' + k + ')"></i>' + label)

    .on('click', function () { off[k] = !off[k]; this.setAttribute('aria-pressed', off[k] ? 'false' : 'true'); render(); });

});


/* ---------- Chart ---------- */

function render() {

  var host = d3.select('#chart'); host.selectAll('*').remove(); tip.style('opacity', 0);

  var W = Math.max(300, document.getElementById('chart').clientWidth), H = W < 560 ? 380 : 460;

  var m = { t: 30, r: 12, b: 52, l: W < 560 ? 44 : 56 }, iw = W - m.l - m.r, ih = H - m.t - m.b;

  var allKeys = d3.range(6).map(function (k) { return 'c' + k; });

  var keys = allKeys.filter(function (k, i) { return !off[i]; });

  var rows = MONTHS.map(function (name, i) { var o = { cat: name }; allKeys.forEach(function (k, c) { o[k] = data[c][i]; }); return o; });


  var tot = d3.sum(data, function (a) { return d3.sum(a); }), jja = d3.sum(data, function (a) { return a[5] + a[6] + a[7]; });

  d3.select('#csub').text(tot + ' cases with a recorded month, 2008 to 2024');


  var svg = host.append('svg').attr('viewBox', '0 0 ' + W + ' ' + H).attr('role', 'img')

    .attr('aria-label', 'Stacked bar chart of recorded deaths by month and cause. June to August is the peak.');

  var g = svg.append('g').attr('transform', 'translate(' + m.l + ',' + m.t + ')');


  var x = d3.scaleBand().domain(MONTHS).range([0, iw]).padding(.22);

  var series = d3.stack().keys(keys)(rows);

  var ymax = d3.max(rows, function (r) { return d3.sum(keys, function (k) { return r[k]; }); }) || 1;

  var y = d3.scaleLinear().domain([0, ymax]).nice().range([ih, 0]);


  // Peak-season band (June to August), drawn first so it sits behind everything

  var bx = x('Jun') - (x.step() - x.bandwidth()) / 2, bw = x.step() * 3;

  g.append('rect').attr('x', bx).attr('y', 0).attr('width', bw).attr('height', ih).style('fill', 'var(--ink)').style('opacity', .06);

  g.append('text').attr('class', 'peak-label').attr('x', bx + bw / 2).attr('y', -10).attr('text-anchor', 'middle')

    .text('June to August: ' + Math.round(jja / tot * 100) + '% of cases');


  g.append('g').call(d3.axisLeft(y).ticks(5).tickSize(-iw).tickFormat(d3.format('d')))

    .call(function (a) { a.select('.domain').remove(); });

  g.append('g').attr('transform', 'translate(0,' + ih + ')').call(d3.axisBottom(x).tickSizeOuter(0));

  svg.append('text').attr('class', 'axis-title').attr('x', m.l + iw / 2).attr('y', H - 8).attr('text-anchor', 'middle').text('Month');

  svg.append('text').attr('class', 'axis-title').attr('transform', 'translate(14,' + (m.t + ih / 2) + ') rotate(-90)').attr('text-anchor', 'middle').text('Recorded deaths');


  if (!keys.length) {

    g.append('text').attr('x', iw / 2).attr('y', ih / 2).attr('text-anchor', 'middle').style('fill', 'var(--muted)').text('Select at least one cause in the legend.');

    return;

  }


  g.append('g').selectAll('g').data(series).join('g').style('fill', function (s) { return 'var(--' + s.key + ')'; })

    .selectAll('rect').data(function (s) { return s.map(function (d) { d.key = s.key; return d; }); }).join('rect')

    .attr('class', 'seg-bar').attr('x', function (d) { return x(d.data.cat); }).attr('width', x.bandwidth())

    .attr('y', ih).attr('height', 0)

    .on('mouseenter mousemove', function (ev, d) {

      var k = +d.key.slice(1), count = d.data[d.key], total = d3.sum(allKeys, function (kk) { return d.data[kk]; });

      var box = wrap.getBoundingClientRect(), left = ev.clientX - box.left + 14, top = ev.clientY - box.top - 70;

      tip.html('<b>' + CAUSES[k] + '</b>' + d.data.cat + ': ' + count + ' case' + (count === 1 ? '' : 's') + '<br>' + (total ? Math.round(count / total * 100) : 0) + '% of cases in this month')

        .style('left', Math.min(left, box.width - 200) + 'px').style('top', Math.max(top, 8) + 'px').style('opacity', 1);

    })

    .on('mouseleave', function () { tip.style('opacity', 0); })

    .transition().duration(550)

    .attr('y', function (d) { return y(d[1]); }).attr('height', function (d) { return Math.max(0, y(d[0]) - y(d[1])); });

}


render();

loadData(0).then(function (d) { data = d; render(); }).catch(function (e) { console.warn('Could not load the CSV. Using the built-in counts.', e); });

var rz; window.addEventListener('resize', function () { clearTimeout(rz); rz = setTimeout(render, 150); });

</script>
